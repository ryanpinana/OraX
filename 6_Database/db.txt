CREATE DATABASE Ora.X; 
USE Ora.X;

CREATE TABLE Utenti (
    Id INTEGER PRIMARY KEY AUTOINCREMENT,
    Nome TEXT NOT NULL,
    Cognome TEXT NOT NULL,
    DataNascita DATE NOT NULL,
    Username TEXT NOT NULL UNIQUE,
    PasswordHash TEXT NOT NULL,
    DataRegistrazione DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE Calendari (
    Id INTEGER PRIMARY KEY AUTOINCREMENT,
    Nome TEXT NOT NULL,
    ProprietarioId INTEGER NOT NULL,
    Condiviso BOOLEAN DEFAULT 0,
    
    FOREIGN KEY (ProprietarioId) REFERENCES Utenti(Id)
);

CREATE TABLE PartecipantiCalendario (
    Id INTEGER PRIMARY KEY AUTOINCREMENT,
    CalendarioId INTEGER NOT NULL,
    UtenteId INTEGER NOT NULL,
    
    FOREIGN KEY (CalendarioId) REFERENCES Calendari(Id),
    FOREIGN KEY (UtenteId) REFERENCES Utenti(Id)
);

CREATE TABLE Categorie (
    Id INTEGER PRIMARY KEY AUTOINCREMENT,
    Nome TEXT NOT NULL,
    Icona TEXT
);

CREATE TABLE Attivita (
    Id INTEGER PRIMARY KEY AUTOINCREMENT,
    CalendarioId INTEGER NOT NULL,
    CreatoreId INTEGER NOT NULL,
    Titolo TEXT NOT NULL,
    Priorita INTEGER,
    Luogo TEXT,
    CategoriaId INTEGER,
    Scadenza DATETIME,
    Note TEXT,
    Completata BOOLEAN DEFAULT 0,
    Ricorrente BOOLEAN DEFAULT 0,
    AttivitaPadreId INTEGER,
    
    FOREIGN KEY (CalendarioId) REFERENCES Calendari(Id),
    FOREIGN KEY (CreatoreId) REFERENCES Utenti(Id),
    FOREIGN KEY (CategoriaId) REFERENCES Categorie(Id),
    FOREIGN KEY (AttivitaPadreId) REFERENCES Attivita(Id)
);

CREATE TABLE InvitiCalendario (
    Id INTEGER PRIMARY KEY AUTOINCREMENT,
    CalendarioId INTEGER NOT NULL,.
    MittenteId INTEGER NOT NULL,
    DestinatarioId INTEGER NOT NULL,
    Stato TEXT DEFAULT 'in_attesa',
    DataInvito DATETIME DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (CalendarioId) REFERENCES Calendari(Id),
    FOREIGN KEY (MittenteId) REFERENCES Utenti(Id),
    FOREIGN KEY (DestinatarioId) REFERENCES Utenti(Id)
);
